const outputFileName = 'project_codebase.txt';

const projectDir = __dirname;

// කේතයට අදාළ නොවන විශාල ෆෝල්ඩර මඟ හැරීම
const excludeDirs = ['.next', 'node_modules', '.git', 'public', '.vercel'];

// කියවිය යුතු ෆයිල් වර්ග (Extensions)
const allowedExtensions = ['.js', '.jsx', '.ts', '.tsx', '.json', '.css', '.md', '.env.local'];

// කලින් සෑදූ ෆයිල් එකක් ඇත්නම් එය මකා දැමීම
if (fs.existsSync(outputFilePath)) {
    fs.unlinkSync(outputFilePath);
}

function processDirectory(directory) {
    const files = fs.readdirSync(directory);

    files.forEach(file => {
        const fullPath = path.join(directory, file);
        const stat = fs.statSync(fullPath);

        if (stat.isDirectory()) {
            if (!excludeDirs.includes(file)) {
                processDirectory(fullPath);
            }
        } else {
            const ext = path.extname(file);
            
            // Output file එක සහ package-lock.json මඟ හැරීම
            if (file === outputFileName || file === 'package-lock.json') return;

            if (allowedExtensions.includes(ext) || file === '.env.local') {
                // Windows පරිගණක වල \ ලකුණ / බවට පත් කර නිවැරදි Path එක සෑදීම
                let relativePath = path.relative(projectDir, fullPath);
                relativePath = relativePath.split(path.sep).join('/'); 
                
                const content = fs.readFileSync(fullPath, 'utf8');
                const separator = `\n\n=== FILE: /${relativePath} ===\n\n`;
                
                fs.appendFileSync(outputFilePath, separator + content);
            }
        }
    });
}

export function getSortedPostsData() {
  const fileNames = fs.readdirSync(postsDirectory);
  const allPostsData = fileNames.map((fileName) => {
    // ෆයිල් එකේ නමින් '.md' කෑලී අයින් කරලා URL එකට ගැළපෙන නම (id) හැදීම
    const slug = fileName.replace(/\.md$/, '');
    const fullPath = path.join(postsDirectory, fileName);
    const fileContents = fs.readFileSync(fullPath, 'utf8');

    // gray-matter මඟින් ලිපියේ Title එක, Date එක වෙන් කිරීම
    const matterResult = matter(fileContents);

    return {
      slug,
      // UI එකට අවශ්‍ය excerpt සහ coverImage ආරක්ෂිතව (safely) ලබා දීම
      excerpt: matterResult.data.description || matterResult.data.excerpt || 'No description available for this article.',
      coverImage: matterResult.data.coverImage || '/images/default-blog-cover.jpg',
      ...matterResult.data,
    };
  });

  // අලුත්ම ලිපිය උඩින්ම පෙන්වීමට දින අනුව පෙළගැස්වීම (Sort by Date)
  return allPostsData.sort((a, b) => {
    if (a.date < b.date) {
      return 1;
    } else {
      return -1;
    }
  });
}

I remember back in 2014 when you could throw some IKEA furniture into a spare bedroom, snap a few photos on an iPhone 4, and watch the bookings roll in. To be honest, those days are long gone. Lately, the biggest question I get from new investors isn’t about which city to buy in; it’s about what *kind* of experience they should be selling. Specifically: "Should I buy a traditional condo/house, or should I go all-in on a glamping site?"

Let’s pull back the curtain on the ROI of glamping versus traditional Airbnbs and see where your money actually works hardest.

## The Traditional Airbnb: The Reliable Workhorse

When we talk about a traditional Airbnb, we’re usually talking about a residential property—a 2-bedroom condo in the city, a suburban 3-bedroom home, or maybe a beach house. These are the bread and butter of the STR industry.

**The Pros of the Traditional Route**
First off, financing is a breeze compared to alternative structures. You can walk into almost any lender and get a conventional mortgage for a single-family home. From an appreciation standpoint, you’re also sitting on a "real" asset. If the Airbnb market tanked tomorrow, you could still sell that house to a family or rent it out long-term to a local tenant.

To be honest, there’s a sense of security in that. You have an exit strategy that doesn’t involve dismantling a geodesic dome.

**The Cons of the Traditional Route**
Here’s the thing: everyone is doing it. Market saturation is a very real threat. In cities like Austin or Phoenix, the sheer volume of "grey-walled, mid-century modern" rentals has driven down the Average Daily Rate (ADR). You’re no longer competing against the hotel down the street; you’re competing against 500 other identical condos. This "commoditization" of the traditional Airbnb means your margins are getting squeezed by rising property taxes, insurance, and management fees.

## Glamping: The High-Yield Rebel

Glamping (glamorous camping) has exploded. We’re talking A-frames, luxury yurts, safari tents, and mirror houses. It’s not just about a bed; it’s about a "bucket list" experience.

**The Pros of the Glamping Route**
The ROI potential in glamping can be astronomical because you are decoupling the price from the square footage. People don't pay $500 a night for a 200-square-foot yurt because it’s a "good deal" on space; they pay it because they want to wake up under a clear canopy and see the stars.

The ADR for a well-executed glamping site often eclipses a 3-bedroom home in the same zip code. Furthermore, your initial "structure" cost (the tent or dome) is often significantly lower than the cost of a stick-built home.

**The Cons of the Glamping Route**
Don't let the Instagram photos fool you—glamping is hard work. To start, permitting is often a nightmare. Many counties don’t even have a zoning code for "luxury tents," which can lead to months of bureaucratic headaches. 

Then there’s the infrastructure. You’re often building on raw land, which means you’re responsible for running power, digging a well, and installing septic systems. And let’s not forget maintenance. A traditional roof lasts 20 years; a canvas safari tent might need replacing in five. Nature is also a brutal landlord—bugs, wind, and temperature fluctuations will be your constant enemies.

## The "Experience Premium" vs. The "Utility Value"

One thing I’ve learned in 10 years as a Superhost is that travelers have two different mindsets. 

1.  **Utility Travelers:** They need a place to stay because they are attending a wedding, a graduation, or a business meeting. They want a kitchen, a reliable Wi-Fi connection, and a comfortable bed. This is where the traditional Airbnb thrives.
2.  **Aspirational Travelers:** They are traveling *for the stay itself*. The property is the destination. This is the glamping crowd.

Here’s the thing: Utility travelers are price-sensitive. They will scroll past your listing for another one that’s $10 cheaper. Aspirational travelers are *not* price-sensitive. If they fall in love with the "vibe" of your glass-walled A-frame, they will pay a premium. This is why glamping often offers a better ROI—you’re selling a feeling, and feelings don't have a price ceiling.

## Operational Realities: Are You a Landlord or an Outfitter?

Before you rush out and buy 10 acres in the desert, we need to talk about the "labor" part of the ROI equation. 

Managing a traditional Airbnb is largely automated. You have a smart lock, a cleaning crew that knows the drill, and a handyman on speed dial. It’s a "passive" investment in the truest sense of the word.

Glamping? Not so much. You’re dealing with specialized equipment. Hot tubs at glamping sites are a massive draw, but they require daily chemical checks. Propane tanks for outdoor heaters need swapping. If a storm rolls through, you might have to check for tears in the canvas or debris on the decks. 

If you’re managing it yourself, your "hourly wage" might actually bring down your real ROI. If you’re hiring a management company, expect to pay 25-35% for glamping, whereas traditional property managers usually charge 15-25%.

## Market Resilience: What Happens in a Recession?

I get asked this a lot. In a downturn, people cut back on luxury travel. Does that mean glamping dies? 

Actually, history suggests the opposite. During the 2008 crash and even the 2020 lockdowns, "drive-to" destinations saw a massive surge. People swapped their $10,000 European vacations for $1,500 long weekends at a cool local spot within a 3-hour drive. 

Traditional Airbnbs in urban centers (think downtown condos) actually suffered more during those times because business travel evaporated. Glamping is the ultimate "staycation" asset. As long as your site is within driving distance of a major metro area, you have a built-in recession hedge.

## Which One Is Right For You?

So, who wins the ROI battle? 

If you are a **passive investor** who wants to leverage traditional financing and sleep soundly knowing your asset will appreciate at 3-5% a year while throwing off a steady 10% return, stick with the **Traditional Airbnb**. It’s the "S&P 500" of the real estate world.

However, if you are an **entrepreneurial investor** who doesn't mind getting your hands a little dirty (or paying a premium for a great manager) and you want to chase 20-30% returns, **Glamping** is the clear winner. It’s the "high-growth tech stock" of real estate.

The traditional house wins here. It’s a liquid asset. Selling a glamping business is more like selling a boutique hotel or a small business. It takes longer and requires a specific type of buyer who understands the P&L (Profit and Loss) statement.

## The Big Misconception: Why Standard Homeowners Insurance Will Fail You

To understand why you need specialized insurance, you need to understand how insurance actuaries price risk.

When you buy a standard Homeowners (HO-3) or Landlord (DP-3) policy, the insurance company assumes the property is occupied by the owner or a long-term tenant who signed a 12-month lease. Long-term occupants know where the light switches are, they know the stairs are a little steep, and they treat the property like a home. 

A short-term rental is entirely different. You have a revolving door of strangers entering a strange environment every three days. They are on vacation. They are drinking. They are trying to use your unfamiliar barbecue grill. 

From an insurance perspective, you are no longer operating a home; **you are operating a decentralized hotel.**

Buried deep in the fine print of almost every standard homeowner's policy is a "Business Activity Exclusion." The moment you hand over the keys in exchange for nightly money, your property becomes a business. If your insurance provider finds out you are running an Airbnb without telling them, they have the legal right to deny any claim—even if the claim had nothing to do with a guest (like a tree falling on the roof).

## "But Airbnb Covers Me!" – The Illusion of AirCover

Whenever the topic of short-term rental insurance comes up in host forums, there is always someone who says, *"I don't need extra insurance. AirCover gives me $3 million in damage protection!"*

This is a very dangerous half-truth. 

Airbnb’s AirCover and VRBO’s Liability Guarantees are fantastic marketing tools, and to be fair, they do pay out for minor things. If a guest ruins your expensive rug with red wine, AirCover will usually reimburse you. 

But you need to understand one critical legal distinction: **AirCover is a host guarantee program, not a commercial insurance policy.** 

Here is why relying solely on platform protections is a massive gamble:

1. **You are not the named insured:** Airbnb holds the policy, not you. They act as the middleman. If things get complicated, their primary goal is to protect their corporate liability, not your personal assets.
2. **Loss of Income is rarely covered:** If a guest burns down your kitchen, AirCover might pay to rebuild the cabinets. But it will take four months to rebuild. ## What Actually IS Short-Term Rental Insurance?

Real **short-term rental insurance** is a comprehensive commercial package policy tailored specifically for the hospitality industry. It replaces your existing homeowner's policy entirely (you don't have two policies, you just have one correct one).

When shopping for a proper policy, it must include these four non-negotiable pillars of coverage:

### 1. Commercial General Liability (CGL)
This is your shield against the big lawsuits. If a guest trips on a loose rug, falls off a balcony, or gets bitten by a dog on your property, they will sue. A commercial liability policy (usually starting at $1,000,000 in coverage) pays for your legal defense team and any settlements, so they can't come after your personal savings or other properties.

### 2. Building and Contents Coverage (Replacement Cost)
This covers the physical structure of your house and everything inside it. *Crucial Tip:* Ensure your policy states **"Replacement Cost"** and not "Actual Cash Value." If a guest steals your 5-year-old leather sofa, Actual Cash Value will give you $100 (what it's worth today at a garage sale). Replacement Cost gives you the $1,500 you need to buy a brand-new one right now.

### 3. Loss of Business Income (Business Interruption)
This is the coverage that saves investors from bankruptcy. If your property becomes unrentable due to a covered peril (like a fire, major water leak, or storm damage), this coverage pays you the gross revenue you *would have made* based on your historical booking data while the home is being repaired. 

### 4. Specialized Hospitality Endorsements
A great short-term rental insurance policy covers the weird things that only happen in this industry:
*   **Bed Bug Extermination:** Covers the cost of pest control and lost income while the property is shut down.
*   **Squatter Protection:** Legal fees to evict a guest who refuses to leave after their booking ends.
*   **Liquor Liability:** Protection if a guest drinks alcohol provided in your welcome basket and then gets into a car accident.

**Cleaning Fee:** $200

**The Old Split-Fee Model (3% Host Fee):**
*   **Total Guest Pays:** $900 (Rent) + $200 (Cleaning) + ~$154 (Guest Fee) = $1,254.
*   **Host Receives:** $1,100 (Subtotal) minus 3% ($33) = **$1,067.**

**The Simplified Pricing Model (15.5% Host Fee) - Without Adjusting Price:**
*   **Total Guest Pays:** $900 (Rent) + $200 (Cleaning) = $1,100.
*   **Host Receives:** $1,100 minus 15.5% ($170.50) = **$929.50.**

**The Difference:** By switching models without changing his base price, this host was losing **$137.50 per booking.** 

If he does 40 bookings a year, that’s **$5,500 in lost revenue** simply because he didn't understand how the fee structure worked. To net the same $1,067, he actually needed to raise his total price to about $1,263. 

## How to Price Correctly (The 1.18x Rule)

So, how do you fix this? You can't just add 15.5% to your price and call it a day. Because you are now paying a fee on the *new, higher price*, the math is a bit more circular.

To be honest, the easiest way to handle this is to use a multiplier. If you want to net what you were netting before, you generally need to multiply your old base rate by **1.18**. 

*   $100 old rate x 1.18 = $118 new rate.
*   $118 - 15.5% ($18.29) = $99.71.

This gets you back to your original baseline. But even this is a "defense" strategy. An expert investor plays offense. 

### 1. Account for Consumables and Wear
With the 15.5% fee, your margin for error is thinner. Every "free" bottle of wine or extra roll of high-end toilet paper is being taxed. I recommend adding a "Consumable Premium" of 2-3% on top of your base rate to cover the fact that Airbnb is taking a cut of your hospitality budget.

### 2. Leverage Dynamic Pricing
You cannot set-it-and-forget-it anymore. Tools like PriceLabs or Wheelhouse are essential, but you must manually set your "Base Price" higher to account for the Simplified Pricing model. If your dynamic pricing software thinks you’re on the 3% model but you’re actually on the 15.5% model, its suggestions will be dangerously low.

### 3. The "Cleaning Fee" Hack
Since Airbnb takes 15.5% of your cleaning fee, you have two choices: 
1.  Increase your cleaning fee by 16% so you aren't paying out of pocket to clean your own house.
2.  Fold the cleaning fee into the nightly rate and advertise "No Cleaning Fee." 

The latter is a power move. If you are already paying a 15.5% fee to be "Simplified," going "No Cleaning Fee" makes your listing incredibly attractive to the Airbnb algorithm and guests alike. 

## The Hidden Opportunity in the 15.5% Fee

It sounds like I’m down on the 15.5% fee, right? Well, not exactly. As an investor, I actually love it—if you know how to use it.

When you opt into Simplified Pricing, your listing gets a "Service Fee: $0" callout. In a sea of listings where guests are getting hit with $200+ in fees at checkout, your listing looks like a bargain, even if your nightly rate is higher. 

The trick is to **market the value.** Use your first five photos to show off the amenities that justify that higher nightly rate. Mention in your description: *"The price you see is the price you pay—no hidden service fees at checkout!"* 

This builds trust. And in the world of short-term rentals, trust is the highest-converting currency.

## The "Other" Hidden Costs You’re Forgetting

While we’re talking about pricing correctly, we have to mention the costs that the 15.5% fee makes more painful:

*   **Occupancy Taxes:** In many jurisdictions, taxes are calculated on the "Gross" amount. If your gross is higher because you’ve baked in the 15.5% fee, your tax liability might slightly shift.
*   **Platform Dependency:** By moving to Simplified Pricing, you are essentially "marrying" Airbnb. It becomes much harder to compare your rates across VRBO or Direct Booking sites because your Airbnb "Base Rate" is now artificially inflated to cover their fee.
*   **Cancellations:** If you have a strict cancellation policy and a guest cancels, Airbnb still takes their cut of the payout. That 15.5% hit on a cancellation payout feels a lot worse than the old 3%.

## FAQs About Airbnb Fees and Pricing

### Is the 15.5% fee mandatory?
Not for everyone. In most of the world (EMEA, Uruguay, etc.), it is mandatory. in the US, Canada, and Mexico, it is mandatory for "Software-Connected" hosts (those using a PMS like Guesty or Hostaway). If you host manually, you can still choose the Split-Fee model, but Airbnb is heavily incentivizing the move to Simplified.

### Does Simplified Pricing really help with SEO?
Airbnb’s algorithm is a "black box," but they have explicitly stated that guests prefer simplified pricing. Listings with no guest service fee tend to have higher conversion rates, and higher conversion rates lead to higher search rankings.

### Should I raise my cleaning fee to cover the Airbnb cut?
Yes. If you don't, you are losing money on every turnover. If your cleaner charges $100, you should charge at least $118 on the platform to ensure you have $100 left after Airbnb's 15.5% take.

### How do I explain the higher nightly rate to returning guests?
Be transparent. Tell them, "We’ve moved to a 'No Guest Fee' model so the price you see is the price you pay. It’s much more transparent!" Most guests won't do the math; they'll just appreciate the lack of "surprise" fees at the end.

## The Bottom Line

Hosting in 2024 and beyond is a game of inches. The days of "accidental" profit are over. To thrive as an Airbnb investor, you have to treat your pricing like a hedge fund manager treats a portfolio. You need to know your "Net Effective Rate"—what stays in your pocket after fees, taxes, and toilet paper.

The 15.5% fee isn't a dealbreaker; it's just a different set of rules. If you price correctly, bake in your expenses, and use the "No Service Fee" badge as a marketing weapon, you can actually out-earn your competitors who are still stuck in the old way of thinking.

## What is Rental Arbitrage?

In simple terms, rental arbitrage is the process of signing a long-term lease for a property and then subletting it as a short-term rental on platforms like Airbnb, Vrbo, or Booking.com.
